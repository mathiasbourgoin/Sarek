type 'fence entry = {
  1. start : int;
  2. fence : 'fence;
}
type 'fence t = {
  1. capacity : int;
  2. align : int;
  3. mutable head : int;
    (*

    Where the next allocation starts.

    *)
  4. live : 'fence entry Stdlib.Queue.t;
    (*

    Oldest first.

    *)
}
val create : capacity:int -> align:int -> 'fence t
val rounded : 'a t -> int -> int
val reclaim : 'a t -> is_done:('a -> bool) -> on_free:('a -> unit) -> unit
val fits : 'a t -> int -> int option
val reserve : 'fence t -> size:int -> is_done:('fence -> bool) -> wait:('fence -> unit) -> on_free:('fence -> unit) -> int

Offset of a free range of size bytes (rounded up to the alignment). Reclaims completed allocations first; when the request still does not fit, calls wait on the oldest live fence (which must complete it), hands it to on_free, and retries. Raises Invalid_argument when size exceeds the capacity or is not positive. The range is NOT reserved until commit.

val commit : 'fence t -> off:int -> size:int -> 'fence -> unit

Mark the size bytes at off live until fence completes. off must come from the reserve immediately before, with the same size.

val live_count : 'fence t -> int

Number of allocations whose fence has not been handed back.
